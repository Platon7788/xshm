cmake_minimum_required(VERSION 4.0)
project(xipc_rust_abi LANGUAGES C)
get_filename_component(_compiler_dir "${CMAKE_C_COMPILER}" DIRECTORY)
find_program(_dumpbin dumpbin HINTS "${_compiler_dir}" REQUIRED)
execute_process(COMMAND "${_dumpbin}" /headers "${XIPC_ARCHIVE}"
    RESULT_VARIABLE _result OUTPUT_VARIABLE _headers ERROR_VARIABLE _error)
if(NOT _result EQUAL 0)
    message(FATAL_ERROR "Cannot inspect xipc.lib architecture: ${_error}")
endif()
string(TOUPPER "${_headers}" _headers)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(_machine "8664")
    set(_foreign_machine "14C")
else()
    set(_machine "14C")
    set(_foreign_machine "8664")
endif()
if(NOT _headers MATCHES "${_machine} MACHINE" OR _headers MATCHES "${_foreign_machine} MACHINE")
    message(FATAL_ERROR "xipc.lib architecture mismatch: install the SDK for this Rust target")
endif()
execute_process(COMMAND "${_dumpbin}" /directives "${XIPC_ARCHIVE}"
    RESULT_VARIABLE _result OUTPUT_VARIABLE _directives ERROR_VARIABLE _error)
if(NOT _result EQUAL 0)
    message(FATAL_ERROR "Cannot inspect xipc.lib CRT: ${_error}")
endif()
string(TOUPPER "${_directives}" _directives)
string(REPLACE "\"" "" _directives "${_directives}")
if(XIPC_RUNTIME STREQUAL "MultiThreaded")
    set(_expected LIBCMT)
    set(_wrong MSVCRT)
else()
    set(_expected MSVCRT)
    set(_wrong LIBCMT)
endif()
if(_directives MATCHES "DEFAULTLIB:${_wrong}" OR
   _directives MATCHES "DEFAULTLIB:LIBCMTD|DEFAULTLIB:MSVCRTD" OR
   NOT _directives MATCHES "DEFAULTLIB:${_expected}([ \r\n]|$)")
    message(FATAL_ERROR
        "xipc.lib CRT mismatch: use a hosted Release SDK built with "
        "XIPC_CRT_FREE=OFF and CMAKE_MSVC_RUNTIME_LIBRARY=${XIPC_RUNTIME}")
endif()
add_library(xipc_rust_abi OBJECT abi_check.c)
target_include_directories(xipc_rust_abi PRIVATE "${XIPC_INCLUDE}")
target_compile_options(xipc_rust_abi PRIVATE /W4 /WX)
set_target_properties(xipc_rust_abi PROPERTIES C_STANDARD 23)
